\section{Rút gọn logic gán nhãn KHTN (diễn giải mã)}
\begin{lstlisting}[language=Python]
# Uu tien theo nhom dieu kien; chon nganh co score uu tien lon nhat
if toan >= 8 and ly >= 8:
    scores[3] = toan + ly + hoa * 0.5   # Ky thuat
if sinh >= 7.5 and hoa >= 7.5:
    scores[2] = sinh + hoa + ly * 0.5 # Y khoa
if toan >= 7.5 and ly >= 7:
    scores[0] = toan + ly + anh * 0.3 # IT
if sinh >= 7 and hoa >= 7:
    scores[4] = sinh + hoa              # Nong - Lam - Ngu
if toan >= 6 and anh >= 7:
    scores[1] = toan + anh + van * 0.3 # Kinh te
if scores:
    return max(scores, key=scores.get)
# ... fallback tong hop neu khong nhanh
\end{lstlisting}

\section{Ghi chú biên dịch \LaTeX}
Dùng \texttt{xelatex} vì gói \texttt{fontspec}. Nếu không có Times New Roman, đổi lệnh \texttt{\textbackslash setmainfont} trong \texttt{main.tex}. Tắt font CJK trùng nếu máy báo lỗi. Với số có dấu phẩy thập phân tiếng Việt, báo cáo dùng dấu \texttt{\{,\}} trong mode toán hoặc viết ``0{,}91'' trong văn bản thường.

\section{Ví dụ chuỗi suy luận (\texttt{chaining\_rules}) --- KHXH}
Trích ý tưởng từ \texttt{rules\_config.json}: sau khi khớp \texttt{KinhTe\_Very\_Fit} hoặc \texttt{KinhTe\_Fit}, nếu Anh $\geq 8$ thì cộng thưởng (\textit{bonus}) cho hướng Kinh tế quốc tế. Các chuỗi tương tự định nghĩa cho Sư phạm, Luật, Du lịch. Việc cộng điểm thưởng giúp phân tầng gợi ý trong cùng một ngành mà không tạo thêm lớp phân loại mới.

\section{Kịch bản kiểm thử thủ công (UAT gợi ý)}
\begin{enumerate}[leftmargin=*]
  \item Nhập vector toàn điểm 10 cho KHTN; kiểm tra thứ tự xếp hạng và tổng điểm lai hợp lý (không âm, không vượt 100 sau clip nội bộ).
  \item Nhập trường hợp một môn trọng tâm rất thấp nhưng các môn khác cao; xác nhận log hoặc giải thích ghi nhận VETO.
  \item Đổi khối KHXH, lặp lại hai bước trên.
  \item Tắt file model (\texttt{.pkl}) tạm thời; xác nhận ứng dụng báo lỗi thân thiện hoặc suy giảm về KBS tùy phiên bản mã.
\end{enumerate}
